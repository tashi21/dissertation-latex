\newtermabbr{adwin}{ADWIN}{adaptive windowing}{A change-detection method that does not fix its window size in advance: the window grows while the data are stable and drops its older part when two sub-windows have sufficiently different means~\cite{Bifet2007}.}
\newtermabbr{birdnest}{BIRDNEST}{Bayesian inference for ratings-fraud detection}{A method that computes a Bayesian estimate of how far the properties of a user, product or rating deviate from those of the whole population. Its normality scores form the behavioural component of REV2~\cite{Kumar2018}.}
\newtermabbr{brs}{BRS}{Beta reputation system}{A Bayesian reputation system that takes binary ratings as input and updates a beta probability distribution. The reputation score is a function of the expected value of that distribution~\cite{Jsang2007}.}
\newtermabbr{cusum}{CUSUM}{cumulative sum}{A sequential-analysis change detector. It accumulates the amount by which successive observations exceed an allowed magnitude of change and signals a change when the sum passes a threshold~\cite{Page1954,Gama2014}.}
\newtermabbr{dgten}{DGTEN}{deep Gaussian-based trust evaluation network}{A graph-based model for dynamic trust evaluation that represents nodes and edges as Gaussian distributions, learns temporal features, and is evaluated against on-off attacks~\cite{Usman2025}.}
\newtermabbr{ks}{KS}{Kolmogorov--Smirnov}{A statistical test of whether two samples originate from the same distribution~\cite{dosReis2016}.}
\newtermabbr{mcp}{MCP}{Model Context Protocol}{An open standard that defines how AI applications discover and call external tools and data sources. An AI application acts as an MCP client and connects to MCP servers, each of which exposes tools through one common interface.}
\newtermabbr{ph}{PH}{Page--Hinkley}{A variant of CUSUM, used for sequential change detection, whose test variable is the cumulative difference between the observed values and their mean~\cite{Gama2014}.}
\newtermabbr{raptar}{RaPTaR}{reacting and predicting in trust and reputation}{A trust-assessment method that deletes old interaction outcomes using a modification of ADWIN and locates the point of behaviour change with a Kolmogorov--Smirnov test on the outcomes of a group of agents~\cite{Player2020}.}

% ========================================== definitions ===========================================
\newglossaryentry{agent}{name={agent}, description={A system built on a large language model that is given access to external tools, with which it can retrieve live data, perform computations and act on external systems~\cite{Hou2026}. In this dissertation, agents may act for clients and rate the servers they use.}}
\newglossaryentry{average-run-length}{name={average run length}, description={The expected number of observations before a monitoring scheme takes action. When nothing has changed it measures the cost of false alarms, and after a change it measures the delay before the change is acted on~\cite{Page1954}.}}
\newglossaryentry{behavioural-component}{name={behavioural component}, description={The part of REV2 that adds BIRDNEST normality scores of users, ratings and products to the fairness, reliability and goodness equations~\cite{Kumar2018}. Left out of the configuration used in this dissertation.}}
\newglossaryentry{benign-fluctuation}{name={benign fluctuation}, description={Change in the trust score of a benign server that is not a malicious shift, for example a temporary dip in quality that recovers, variation from having few ratings, or an effect of windowing. The delta threshold has to tolerate it.}}
\newglossaryentry{cold-start}{name={cold start}, description={The situation in which a user or product has given or received too few ratings to be judged reliably. REV2 addresses it with Laplace smoothing~\cite{Kumar2018}.}}
\newglossaryentry{concept-drift}{name={concept drift}, description={A change over time in the data a model learns from, in particular in the relationship between its inputs and the target it predicts. A drift may be sudden, incremental or gradual, and a concept may reoccur~\cite{Gama2014}.}}
\newglossaryentry{cross-server-shadowing}{name={cross-server shadowing}, description={An attack in which one of several MCP servers installed together imitates the tools of a trusted server, so that the trusted tool's invocation triggers the malicious one~\cite{Hou2026}.}}
\newglossaryentry{delta-threshold}{name={delta threshold}, description={The magnitude of divergence between a server's historical and recent trust scores beyond which the windowed mechanism reports the recent score in place of the historical one. Its calibration is the subject of the sub-question.}}
\newglossaryentry{detection-delay}{name={detection delay}, description={The number of observations, or the time, needed to detect a change after it has occurred~\cite{Gama2014}.}}
\newglossaryentry{ecological-validity}{name={ecological validity}, description={The extent to which the results of a study carry over to real-world settings.}}
\newglossaryentry{fairness}{name={fairness}, description={In REV2, the trustworthiness of a user as a rater, from 0 (completely untrustworthy) to 1 (completely trustworthy)~\cite{Kumar2018}. Unrelated to fairness in machine learning.}}
\newglossaryentry{false-alarm}{name={false alarm}, description={A signal that a change has occurred when none has. For the mechanism studied here, a benign server that is wrongly flagged. Also called a false positive.}}
\newglossaryentry{forgetting-factor}{name={forgetting factor}, description={A factor by which past ratings are discounted, so that older evidence counts for less. Similar factors are called aging or fading factors~\cite{Jsang2007}.}}
\newglossaryentry{goodness}{name={goodness}, description={In REV2, the quality of a product: a single number from $-1$ to $+1$ indicating the most likely rating a fair user would give it~\cite{Kumar2018}. In this dissertation, the trust score of an MCP server.}}
\newglossaryentry{gradual-poisoning}{name={gradual poisoning}, description={A stress condition in which a server's quality declines slowly after the attack begins, instead of dropping at once as in a rug pull.}}
\newglossaryentry{hellinger-distance}{name={Hellinger distance}, description={A measure of the distance between two probability distributions. Used in drift detection between histograms of current data and of a reference distribution~\cite{Ditzler2011}.}}
\newglossaryentry{historical-window}{name={historical window}, description={The earlier of the two time periods of the rating graph that the windowed mechanism scores separately with REV2. Its goodness scores are compared with those of the recent window.}}
\newglossaryentry{landmark-window}{name={landmark window}, description={A window that holds everything that has arrived since a given time~\cite{Gama2014}.}}
\newglossaryentry{laplace-smoothing}{name={Laplace smoothing}, description={Adding pseudocounts at a prior value to the real ratings of a user or product, so that a score computed from few ratings stays close to the prior and moves away from it as ratings accumulate~\cite{Kumar2018}.}}
\newglossaryentry{mcp-client}{name={MCP client}, text={client}, description={The component, run by the host, that keeps a one-to-one connection with a single MCP server, queries the functions that server offers and sends requests to it on behalf of the host~\cite{Hou2026}.}}
\newglossaryentry{mcp-host}{name={MCP host}, text={host}, description={The AI application in which tasks are carried out, such as Claude Desktop, Codex or an autonomous agent. It runs the MCP client~\cite{Hou2026}.}}
\newglossaryentry{mcp-market}{name={MCP market}, text={market}, description={A platform on which MCP servers are published for users to find and install. Also called a server registry. Most are run by third parties~\cite{Hou2026}.}}
\newglossaryentry{mcp-server}{name={MCP server}, text={server}, description={A program that offers tools, resources and prompts to an MCP client~\cite{Hou2026}. In the mapping used in this dissertation, a server takes the role of a REV2 product.}}
\newglossaryentry{mcp-tool}{name={MCP tool}, text={tool}, description={A function offered by an MCP server that carries out an operation through external services and APIs~\cite{Hou2026}.}}
\newglossaryentry{on-off-attack}{name={on-off attack}, description={An attack in which a malicious node behaves well and badly alternately, so that a trust system treats its bad behaviour as a temporary error~\cite{Chae2015}. Also called a cyclic attack: the attacker repeatedly regains high trust through cooperation and then abuses it~\cite{Salehi-Abari2010}.}}
\newglossaryentry{one-way-shift}{name={one-way shift}, description={A change of behaviour in one direction only: a server that behaves benignly and then turns malicious, as in a rug pull. Contrasts with an on-off attack, which alternates.}}
\newglossaryentry{prior}{name={prior}, description={A belief about a score held before any ratings have been seen. In REV2, the default fairness of a user that has given no ratings and the default goodness of a product that has received none~\cite{Kumar2018}.}}
\newglossaryentry{pseudocount}{name={pseudocount}, description={A non-negative integer number of imaginary ratings that Laplace smoothing adds to the real ratings of a user or product~\cite{Kumar2018}.}}
\newglossaryentry{puppet-attack}{name={puppet attack}, description={An attack in which one of several MCP servers installed together manipulates the tools of a trusted server, so that the trusted tool's invocation triggers the malicious one~\cite{Song2026}.}}
\newglossaryentry{rating-graph}{name={rating graph}, description={A directed bipartite graph of users and products in which each edge is a timestamped rating from a user to a product~\cite{Kumar2018}.}}
\newglossaryentry{recent-window}{name={recent window}, description={The later of the two time periods of the rating graph that the windowed mechanism scores separately with REV2, holding the most recent ratings. Its goodness scores are compared with those of the historical window.}}
\newglossaryentry{redemption}{name={redemption}, description={A mechanism by which a trust system lets a node recover trust after temporary errors, based on later good behaviour or on the passing of time~\cite{Chae2015}.}}
\newglossaryentry{reliability}{name={reliability}, description={In REV2, the trustworthiness of a single rating, from 0 (untrustworthy) to 1 (trustworthy)~\cite{Kumar2018}. Unrelated to reliability in statistics.}}
\newglossaryentry{reputation}{name={reputation}, description={What is generally said or believed about the character or standing of a person or thing~\cite{Jsang2007}.}}
\newglossaryentry{reputation-system}{name={reputation system}, description={A system in which users rate a provider after an interaction, and the aggregated ratings are used to derive a trust or reputation score for that provider~\cite{Jsang2007}.}}
\newglossaryentry{rev2}{name={REV2}, description={A rating-graph algorithm that scores the fairness of users, the reliability of ratings and the goodness of products jointly, from the timestamped ratings in the graph. Introduced to predict fraudulent users~\cite{Kumar2018}.}}
\newglossaryentry{rug-pull}{name={rug pull}, description={An attack in which an MCP server that behaved as advertised when first published is later altered by its maintainer to introduce malicious code or end its safe functionality~\cite{Hou2026}.}}
\newglossaryentry{sliding-window}{name={sliding window}, description={A window that stores only the most recent examples and moves forward as new ones arrive~\cite{Gama2014}.}}
\newglossaryentry{time-window}{name={time window}, description={A window defined by a duration rather than a number of observations: the set of ratings whose timestamps fall within a given period. Also called a timestamp-based window~\cite{Gama2014}.}}
\newglossaryentry{tool-poisoning}{name={tool poisoning}, description={An attack in which instructions hidden in the description of an MCP tool, which the user cannot see, lead the model to leak data or take unauthorised actions while the tool still returns correct results~\cite{Song2026}.}}
\newglossaryentry{traitor}{name={traitor}, description={An attacker that behaves honestly for an initial period to build up a positive reputation before starting a self-serving attack~\cite{Hoffman2009}.}}
\newglossaryentry{trust}{name={trust}, description={How far one party is willing to depend on something or somebody in a given situation, with a feeling of relative security, even though negative consequences are possible~\cite{Jsang2007}.}}
\newglossaryentry{trust-score}{name={trust score}, description={The value that expresses how far a server can be trusted. In this dissertation, a server's REV2 goodness.}}
\newglossaryentry{trustee}{name={trustee}, description={The agent whose trustworthiness is assessed by another agent, usually from its past behaviour~\cite{Parhizkar2022}.}}
