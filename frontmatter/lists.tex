% ==================================================================================================
% frontmatter/lists.tex: table of contents, then the lists of tables, figures, and algorithms.
%
% The lists are generated from \caption commands (tables, figures, algorithms) and headings
% (contents), via auxiliary files written on the previous compile run, so new items appear after
% the next run.
% ==================================================================================================

\clearpage              % the contents start on a new page after the abstract

\begingroup             % start a group: the changes below stay inside it
\setstretch{1}          % lists are single-spaced even though body text is 1.5
\let\clearpage\relax    % each list heading is an unnumbered \chapter*, which normally starts a new
                        % page; disabling \clearpage inside this group keeps the lists on the same
                        % page. Chapter 1 still starts a new page because \mainmatter comes after
                        % the group ends.

\tableofcontents        % "Contents"; not listed in itself

% Each block below:
%   \vspace{...}                            gap above the next list heading (chapter headings have
%                                           no space above them)
%   \phantomsection                         invisible anchor so the ToC link jumps to this spot
%   \addcontentsline{toc}{chapter}{title}   add a chapter-level ToC entry
%   \listof...                              print the list

\vspace{2\baselineskip}\phantomsection
\addcontentsline{toc}{chapter}{\listtablename}      % "List of Tables"
\listoftables

\vspace{2\baselineskip}\phantomsection
\addcontentsline{toc}{chapter}{\listfigurename}     % "List of Figures"
\listoffigures

\vspace{2\baselineskip}\phantomsection
\addcontentsline{toc}{chapter}{\listalgorithmname}  % "List of Algorithms"
\listofalgorithms                                   % from the algorithm package

\endgroup   % back to normal spacing and page breaks
