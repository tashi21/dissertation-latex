% ==================================================================================================
% preamble/fonts.tex: document fonts. Compile with XeLaTeX.
%
% Aptos is loaded from the font files in assets/font-aptos/. Maths stays in the default Latin
% Modern maths font.
% ==================================================================================================

\usepackage{fontspec}

% settings shared by every font loaded below
\defaultfontfeatures{
  Path           = assets/font-aptos/,
  Extension      = .ttf,
  UprightFont    = *,
  BoldFont       = *-Bold,
  ItalicFont     = *-Italic,
  BoldItalicFont = *-Bold-Italic,
}

\setmainfont{Aptos}  % body font
\setsansfont{Aptos}  % sans-serif font
