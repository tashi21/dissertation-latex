% ==================================================================================================
% preamble/fonts.tex: document fonts. Compile with XeLaTeX.
%
% Aptos is loaded from the font files in assets/font-aptos/
% Path: folder holding the files, relative to main.tex. Extension: added to each file name.
% Maths stays in the default Latin Modern maths font.
% ==================================================================================================

\usepackage{fontspec}
% Sets the body font
\setmainfont{Aptos}[
  Path           = assets/font-aptos/,
  Extension      = .ttf,
  UprightFont    = *,   % "*" stands for the base name "Aptos"
  BoldFont       = *-Bold,
  ItalicFont     = *-Italic,
  BoldItalicFont = *-Bold-Italic,
]
% Sets the sans-serif font to the same as the body font
\setsansfont{Aptos}[
  Path           = assets/font-aptos/,
  Extension      = .ttf,
  UprightFont    = *,
  BoldFont       = *-Bold,
  ItalicFont     = *-Italic,
  BoldItalicFont = *-Bold-Italic,
]
